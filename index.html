// ===== CMS DATA FETCH & RENDER LOGIC ===== async function loadCMSContent() { try { // 1. Fetch and Render About Data const aboutResponse = await fetch('about.json'); if (aboutResponse.ok) { const aboutData = await aboutResponse.json(); if (aboutData.bioTag) { const bioTagEl = document.querySelector('.bio-tag'); if (bioTagEl) bioTagEl.textContent = aboutData.bioTag; } if (aboutData.bioColumns && Array.isArray(aboutData.bioColumns)) { const bioColsContainer = document.querySelector('.bio-columns'); if (bioColsContainer) { bioColsContainer.innerHTML = aboutData.bioColumns .map(col => `
${col}
`).join(''); } } } // 2. Fetch and Render Concerts Data const concertsResponse = await fetch('concerts.json'); if (concertsResponse.ok) { const concertsData = await concertsResponse.json(); if (concertsData.datesFooter) { const datesFooterEl = document.querySelector('.dates-footer'); if (datesFooterEl) datesFooterEl.innerHTML = concertsData.datesFooter; } if (concertsData.concerts && Array.isArray(concertsData.concerts)) { const datesListEl = document.querySelector('.dates-list'); if (datesListEl) { datesListEl.innerHTML = ''; const today = new Date(); today.setHours(0, 0, 0, 0); concertsData.concerts.forEach(item => { const row = document.createElement('div'); row.className = 'date-row'; if (item.date) { const concertDate = new Date(item.date + 'T00:00:00'); if (concertDate < today) row.classList.add('past-date'); } let dateDisplay = item.displayDate; if (!dateDisplay && item.date) { const d = new Date(item.date + 'T00:00:00'); dateDisplay = `${d.getDate()} ${d.toLocaleString('en-GB', { month: 'short' })} '${d.getFullYear().toString().slice(-2)}`; } row.innerHTML = `${dateDisplay || ''}${item.event || ''}`; datesListEl.appendChild(row); }); } } } } catch (err) { console.error('Error fetching CMS content:', err); } }
